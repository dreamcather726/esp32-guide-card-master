cmake_minimum_required(VERSION 3.16)
project(user_binding_tests LANGUAGES C CXX)
enable_testing()
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(CJSON_SOURCE_DIR "$ENV{IDF_PATH}/components/json/cJSON" CACHE PATH "Directory containing cJSON.c and cJSON.h")
if(NOT EXISTS "${CJSON_SOURCE_DIR}/cJSON.c")
    message(FATAL_ERROR "Set CJSON_SOURCE_DIR to the ESP-IDF cJSON source directory")
endif()
add_library(binding_test_cjson STATIC "${CJSON_SOURCE_DIR}/cJSON.c")
target_include_directories(binding_test_cjson PUBLIC "${CJSON_SOURCE_DIR}")
foreach(test_name user_binding_test user_binding_json_test)
    add_executable(${test_name} "${test_name}.cc")
    target_include_directories(${test_name} PRIVATE "../main/boards/common")
    target_link_libraries(${test_name} PRIVATE Threads::Threads binding_test_cjson)
    add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()
